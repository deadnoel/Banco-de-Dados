 -- FUNcoes de consulta
/*AVG > media
SUM () > soma
min menor valor
max maior valor
count() retorna o numero de linhas ou registros encontrados*/

-- QTT total de consultas
 
 SELECT COUNT(*) AS total_consultas
 FROM Consultas;

 SELECT MAX (valor) AS maior_valor
 FROM Consultas;

 SELECT MIN (valor) AS menor_valor
 FROM Consultas;

 SELECT AVG (valor) AS media_valor
 FROM Consultas;

 SELECT SUM (valor) AS soma_valor
 FROM Consultas;

 -- GROUP - Agrupar Resultados
 --contar quantos pacientes tem convenio
 SELECT convenio, COUNT(*) AS total_paciente
 FROM Pacientes
 GROUP BY convenio;
 --Média de valores ce consultas por paciente
 SELECT id_paciente, AVG(valor) AS media_valor
 FROM Consultas
 GROUP BY id_paciente;
